const SUPABASE_URL = "https://oekljkhftgcvuhepmxua.supabase.co";
const SUPABASE_ANON_KEY = "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6Im9la2xqa2hmdGdjdnVoZXBteHVhIiwicm9sZSI6ImFub24iLCJpYXQiOjE3ODg1MzE3NDEsImV4cCI6MjEwNDEwNzc0MX0.5UabBvkHpeTOb756ZQHdyTPd9b_5kz3RusWF-xiSaw0";

const CACHE_NAME = 'tri-putra-abadi-v5-secure';
const ASSETS = [
  './',
  './index.html',
  './manifest.json',
  './icon-192.png',
  './icon-512.png',
  './icon-bulat-1024.png',
  './logo-full.png',
  './maskable-512.png'
];

// Expose config ke window via postMessage
self.addEventListener('install', e => {
  e.waitUntil(caches.open(CACHE_NAME).then(c => c.addAll(ASSETS)));
  self.skipWaiting();
});

self.addEventListener('activate', e => {
  e.waitUntil(caches.keys().then(keys => Promise.all(keys.filter(k => k !== CACHE_NAME).map(k => caches.delete(k)))));
  self.clients.claim();
});

self.addEventListener('fetch', e => {
  // Bypass supabase & pakasir api calls - jangan di-cache
  if (e.request.url.includes('supabase.co') || e.request.url.includes('pakasir')) {
    return;
  }
  e.respondWith(
    caches.match(e.request).then(r => r || fetch(e.request).then(fetchRes => {
      return caches.open(CACHE_NAME).then(cache => {
        cache.put(e.request, fetchRes.clone());
        return fetchRes;
      });
    })).catch(()=> caches.match('./index.html'))
  );
});

// Endpoint untuk ambil config aman dari SW (dipanggil index.html)
self.addEventListener('message', event => {
  if (event.data && event.data.type === 'GET_SUPABASE_CONFIG') {
    event.ports[0].postMessage({
      url: SUPABASE_URL,
      anonKey: SUPABASE_ANON_KEY
    });
  }
});
